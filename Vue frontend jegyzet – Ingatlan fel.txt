6. Route-ok beállítása

src/router/index.js

import { createRouter, createWebHistory } from 'vue-router'

import OpenPage from '../views/OpenPage.vue'
import OffersView from '../views/OffersView.vue'
import NewAdView from '../views/NewAdView.vue'

const router = createRouter({
  history: createWebHistory(),
  routes: [
    {
      path: '/',
      name: 'openpage',
      component: OpenPage
    },
    {
      path: '/offers',
      name: 'offers',
      component: OffersView
    },
    {
      path: '/newad',
      name: 'newad',
      component: NewAdView
    }
  ]
})

export default router

Kötelező útvonalak:

/        kezdőoldal
/offers  hirdetések listája
/newad   új hirdetés feladása

9. Hirdetések megjelenítése komponens

Ez kéri le:

GET http://localhost:5000/api/ingatlan

src/views/OffersView.vue

<template>
  <div class="container my-5">
    <h1 class="mb-4">Aktuális ajánlataink</h1>

    <RouterLink to="/" class="btn btn-secondary mb-3">
      Vissza a főoldalra
    </RouterLink>

    <table class="table table-striped table-bordered align-middle">
      <thead class="table-dark">
        <tr>
          <th>Kategória</th>
          <th>Leírás</th>
          <th>Hirdetés dátuma</th>
          <th>Tehermentes</th>
          <th>Kép</th>
        </tr>
      </thead>

      <tbody>
        <tr v-for="ingatlan in ingatlanok" :key="ingatlan.id">
          <td>{{ ingatlan.kategoriaNev }}</td>
          <td>{{ ingatlan.leiras }}</td>
          <td>{{ ingatlan.hirdetesDatuma }}</td>
          <td>
            <span v-if="ingatlan.tehermentes" class="badge bg-success">
              Igen
            </span>
            <span v-else class="badge bg-danger">
              Nem
            </span>
          </td>
          <td>
            <img
              :src="ingatlan.kepUrl"
              alt="Ingatlan képe"
              class="property-img"
            >
          </td>
        </tr>
      </tbody>
    </table>

    <div v-if="hiba" class="alert alert-danger mt-3">
      {{ hiba }}
    </div>
  </div>
</template>

<script>
import axios from 'axios'

export default {
  name: 'OffersView',

  data() {
    return {
      ingatlanok: [],
      hiba: ''
    }
  },

  mounted() {
    this.ingatlanokLekerese()
  },

  methods: {
    async ingatlanokLekerese() {
      try {
        const response = await axios.get('http://localhost:5000/api/ingatlan')
        this.ingatlanok = response.data
      } catch (error) {
        this.hiba = 'Hiba történt az ingatlanok lekérésekor.'
      }
    }
  }
}
</script>

<style scoped>
.property-img {
  width: 120px;
  height: 80px;
  object-fit: cover;
}
</style>
10. Új hirdetés komponens

Ez kell hozzá:

GET  /api/kategoriak
POST /api/ujingatlan

src/views/NewAdView.vue

<template>
  <div class="container my-5">
    <h1 class="mb-4">Új hirdetés feladása</h1>

    <RouterLink to="/" class="btn btn-secondary mb-3">
      Vissza a főoldalra
    </RouterLink>

    <form @submit.prevent="hirdetesFeladasa" class="card p-4">
      <div class="mb-3">
        <label class="form-label">Ingatlan típusa</label>
        <select
          v-model="ujIngatlan.kategoriaId"
          class="form-select"
          required
        >
          <option value="">Válasszon kategóriát</option>
          <option
            v-for="kategoria in kategoriak"
            :key="kategoria.id"
            :value="kategoria.id"
          >
            {{ kategoria.megnevezes }}
          </option>
        </select>
      </div>

      <div class="mb-3">
        <label class="form-label">Leírás</label>
        <textarea
          v-model="ujIngatlan.leiras"
          class="form-control"
          rows="4"
          required
        ></textarea>
      </div>

      <div class="mb-3">
        <label class="form-label">Hirdetés dátuma</label>
        <input
          v-model="ujIngatlan.hirdetesDatuma"
          type="date"
          class="form-control"
          readonly
        >
      </div>

      <div class="mb-3 form-check">
        <input
          v-model="ujIngatlan.tehermentes"
          type="checkbox"
          class="form-check-input"
          id="tehermentes"
        >
        <label class="form-check-label" for="tehermentes">
          Tehermentes
        </label>
      </div>

      <div class="mb-3">
        <label class="form-label">Kép URL</label>
        <input
          v-model="ujIngatlan.kepUrl"
          type="url"
          class="form-control"
          required
        >
      </div>

      <button type="submit" class="btn btn-success">
        Hirdetés feladása
      </button>
    </form>

    <div v-if="hiba" class="alert alert-danger mt-3">
      {{ hiba }}
    </div>
  </div>
</template>

<script>
import axios from 'axios'

export default {
  name: 'NewAdView',

  data() {
    return {
      kategoriak: [],
      hiba: '',
      ujIngatlan: {
        kategoriaId: '',
        leiras: '',
        hirdetesDatuma: '',
        tehermentes: false,
        kepUrl: ''
      }
    }
  },

  mounted() {
    this.maiDatumBeallitasa()
    this.kategoriakLekerese()
  },

  methods: {
    maiDatumBeallitasa() {
      const today = new Date()
      this.ujIngatlan.hirdetesDatuma = today.toISOString().split('T')[0]
    },

    async kategoriakLekerese() {
      try {
        const response = await axios.get('http://localhost:5000/api/kategoriak')
        this.kategoriak = response.data
      } catch (error) {
        this.hiba = 'Hiba történt a kategóriák lekérésekor.'
      }
    },

    async hirdetesFeladasa() {
      try {
        await axios.post(
          'http://localhost:5000/api/ujingatlan',
          this.ujIngatlan
        )

        this.$router.push('/offers')
      } catch (error) {
        this.hiba = 'Hiba történt a hirdetés feladása során.'
      }
    }
  }
}
</script>
11. Nagyon fontos dátum formátum

A POST kérésben ez kell:

"hirdetesDatuma": "2022-03-09"

Ezért használjuk:

today.toISOString().split('T')[0]

Ez ilyen formátumot ad:

YYYY-MM-DD

16. Gyakori hibák
CORS hiba

Ha ezt látod:

Access to XMLHttpRequest has been blocked by CORS policy

A backend nem engedi a frontend hívást. Vizsgás API-nál általában be van állítva, de ha nem, másik port vagy proxy kellhet.

Nem jelennek meg az adatok

Ellenőrizd:

console.log(response.data)

A lekérés után:

const response = await axios.get('http://localhost:5000/api/ingatlan')
console.log(response.data)
Rossz mezőnevek

A szerver ilyen mezőket ad:

ingatlan.kategoriaNev
ingatlan.leiras
ingatlan.hirdetesDatuma
ingatlan.tehermentes
ingatlan.kepUrl

Ne ezt írd:

ingatlan.kategoria
ingatlan.kep
ingatlan.datum
Select nem működik

A kategóriák mezőnevei:

kategoria.id
kategoria.megnevezes

Nem:

kategoria.nev
POST nem működik

Ellenőrizd, hogy pontosan ezek a mezők mennek-e:

{
  kategoriaId: 1,
  leiras: '...',
  hirdetesDatuma: '2022-03-09',
  tehermentes: true,
  kepUrl: '...'
}
