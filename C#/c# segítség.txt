5. Ad osztály

Ad.cs

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;

class Ad
{
    public int Id { get; set; }
    public string Description { get; set; }
    public int Rooms { get; set; }
    public int Area { get; set; }
    public int Floors { get; set; }
    public Category Category { get; set; }
    public Seller Seller { get; set; }
    public bool FreeOfCharge { get; set; }
    public string ImageUrl { get; set; }
    public double Lat { get; set; }
    public double Long { get; set; }
    public DateTime CreateAt { get; set; }

    public Ad(string sor)
    {
        string[] adatok = sor.Split(';');

        Id = int.Parse(adatok[0]);
        Description = adatok[1];
        Rooms = int.Parse(adatok[2]);
        Area = int.Parse(adatok[3]);
        Floors = int.Parse(adatok[4]);

        Category = new Category(
            int.Parse(adatok[5]),
            adatok[6]
        );

        Seller = new Seller(
            int.Parse(adatok[7]),
            adatok[8],
            adatok[9]
        );

        FreeOfCharge = bool.Parse(adatok[10]);
        ImageUrl = adatok[11];

        string[] koordinatak = adatok[12].Split(',');
        Lat = double.Parse(koordinatak[0], CultureInfo.InvariantCulture);
        Long = double.Parse(koordinatak[1], CultureInfo.InvariantCulture);

        CreateAt = DateTime.Parse(adatok[13]);
    }

    public static List<Ad> LoadFromCsv(string fajlnev)
    {
        return File.ReadAllLines(fajlnev)
            .Skip(1)
            .Select(sor => new Ad(sor))
            .ToList();
    }

    public double DistanceTo(double lat, double lon)
    {
        double x = Lat - lat;
        double y = Long - lon;

        return Math.Sqrt(x * x + y * y);
    }
}

7. CSV beolvasás

A feladat kéri, hogy legyen statikus metódus:

public static List<Ad> LoadFromCsv(string fajlnev)

Használata:

List<Ad> hirdetesek = Ad.LoadFromCsv("realestates.csv");

Ez:

beolvassa a fájlt
kihagyja az első sort
minden sorból Ad objektumot készít
listát ad vissza
8. Program.cs alap szerkezete

Program.cs

using System;
using System.Collections.Generic;
using System.Linq;

namespace RealEstate
{
    class Program
    {
        static void Main(string[] args)
        {
            List<Ad> hirdetesek = Ad.LoadFromCsv("realestates.csv");

            Console.WriteLine("6. feladat:");
            double atlag = hirdetesek
                .Where(x => x.Floors == 0)
                .Average(x => x.Area);

            Console.WriteLine($"Földszinti ingatlanok átlagos alapterülete: {atlag:F2} m2");

            Console.WriteLine();

            Console.WriteLine("8. feladat:");
            double mesevarLat = 47.4164220114023;
            double mesevarLong = 19.066342425796986;

            Ad legkozelebbi = hirdetesek
                .Where(x => x.FreeOfCharge)
                .OrderBy(x => x.DistanceTo(mesevarLat, mesevarLong))
                .First();

            Console.WriteLine("Mesevár óvodához légvonalban legközelebbi tehermentes ingatlan adatai:");
            Console.WriteLine($"\tEladó neve     : {legkozelebbi.Seller.Name}");
            Console.WriteLine($"\tEladó telefonja: {legkozelebbi.Seller.Phone}");
            Console.WriteLine($"\tAlapterület    : {legkozelebbi.Area} m2");
            Console.WriteLine($"\tSzobák száma   : {legkozelebbi.Rooms}");

            Console.ReadKey();
        }
    }
}

10. 7. feladat – DistanceTo metódus

Az Ad osztályba kell írni:

public double DistanceTo(double lat, double lon)
{
    double x = Lat - lat;
    double y = Long - lon;

    return Math.Sqrt(x * x + y * y);
}

Ez Pitagorasz-tétellel számol távolságot:

távolság = √((lat1 - lat2)² + (long1 - long2)²)

17. Gyakori hibák
Nem találja a CSV fájlt

Megoldás:

realestates.csv -> Properties
Copy to Output Directory -> Copy if newer
Hibás double parse

Ha a koordináta ponttal van írva:

47.4164220114023

Akkor kell:

CultureInfo.InvariantCulture

Példa:

double.Parse(koordinatak[0], CultureInfo.InvariantCulture)
A fejlécsort is beolvassa

Ez hibát okozhat:

File.ReadAllLines(fajlnev)

Ez a jó:

File.ReadAllLines(fajlnev).Skip(1)
Rossz logikai érték

Ha a CSV-ben ez van:

true
false

Akkor jó:

bool.Parse(adatok[10])

Ha ez van:

1
0

Akkor ezt kell használni:

FreeOfCharge = adatok[10] == "1";
