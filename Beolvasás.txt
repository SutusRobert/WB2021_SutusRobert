7. Txt beolvasás és összegzés

Most legyen ilyen a fájl:

Alma;300;2
Körte;250;4
Banán;500;3

Jelentés:

termék neve
ár
darab

Feladat: teljes összeg kiszámítása.

private void buttonOsszeg_Click(object sender, EventArgs e)
{
    string[] sorok = File.ReadAllLines("termekek.txt");
    int vegosszeg = 0;

    foreach (string sor in sorok)
    {
        string[] adatok = sor.Split(';');

        if (adatok.Length == 3)
        {
            int ar = int.Parse(adatok[1]);
            int db = int.Parse(adatok[2]);

            vegosszeg += ar * db;
        }
    }

    label1.Text = "Végösszeg: " + vegosszeg + " Ft";
}

9. Txt-ből betöltés listákba, utána lekérdezés

Ez már kicsit komolyabb, és dolgozatra jó.

List<string> nevek = new List<string>();
List<int> korok = new List<int>();
List<string> varosok = new List<string>();

private void buttonBetolt_Click(object sender, EventArgs e)
{
    nevek.Clear();
    korok.Clear();
    varosok.Clear();

    string[] sorok = File.ReadAllLines("adatok.txt");

    foreach (string sor in sorok)
    {
        string[] adatok = sor.Split(';');

        if (adatok.Length == 3)
        {
            nevek.Add(adatok[0]);
            korok.Add(int.Parse(adatok[1]));
            varosok.Add(adatok[2]);
        }
    }

    MessageBox.Show("Beolvasás kész!");
}

Lekérdezés például név alapján:

private void buttonKeres_Click(object sender, EventArgs e)
{
    listBox1.Items.Clear();

    for (int i = 0; i < nevek.Count; i++)
    {
        if (nevek[i].ToLower().Contains(textBoxKeres.Text.ToLower()))
        {
            listBox1.Items.Add(nevek[i] + " - " + korok[i] + " - " + varosok[i]);
        }
    }
}
