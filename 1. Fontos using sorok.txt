private void AdatokBetoltese()
{
    string connectionString = @"Data Source=.\SQLEXPRESS;Initial Catalog=AdatbazisNev;Integrated Security=True";
    string query = "SELECT * FROM Tanulok";

    using (SqlConnection conn = new SqlConnection(connectionString))
    {
        try
        {
            conn.Open();

            DataTable dt = new DataTable();
            adapter.Fill(dt);

            dataGridView1.DataSource = dt;
        }
        catch (Exception ex)

        }

}

4. Lekérdezés feltétellel (WHERE)
Például textBox1 alapján keresel névre.


private void Kereses()
{
    string connectionString = @"Data Source=.\SQLEXPRESS;Initial Catalog=AdatbazisNev;Integrated Security=True";
    string query = "SELECT * FROM Tanulok WHERE Nev LIKE @nev";

    using (SqlConnection conn = new SqlConnection(connectionString))
    {
        try
        {
            conn.Open();

            SqlCommand cmd = new SqlCommand(query, conn);
            cmd.Parameters.AddWithValue("@nev", "%" + textBox1.Text + "%");

            SqlDataAdapter adapter = new SqlDataAdapter(cmd);
            DataTable dt = new DataTable();
            adapter.Fill(dt);

            dataGridView1.DataSource = dt;
        }
        catch (Exception ex)
        {
            MessageBox.Show("Hiba: " + ex.Message);
        }
    }
}

Gombhoz:

private void buttonKereses_Click(object sender, EventArgs e)
{
    Kereses();
}

List<string> nevek = new List<string>();List<int> arak = new List<int>();private void buttonBeolvas_Click(object sender, EventArgs e){    OpenFileDialog ofd = new OpenFileDialog();    ofd.Filter = "Text fájl|*.txt";    if (ofd.ShowDialog() == DialogResult.OK)    {        string[] sorok = File.ReadAllLines(ofd.FileName);        nevek.Clear();        arak.Clear();        foreach (string sor in sorok)        {            string[] adat = sor.Split(';');            if (adat.Length >= 2)            {                nevek.Add(adat[0]);                arak.Add(int.Parse(adat[1]));            }        }        listBox1.Items.Clear();        for (int i = 0; i < nevek.Count; i++)        {            listBox1.Items.Add(nevek[i] + " - " + arak[i] + " Ft");        }    }}
