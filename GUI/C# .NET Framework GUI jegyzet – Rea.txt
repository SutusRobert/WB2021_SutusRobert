6. Seller osztály

Ha nem hivatkozod be a konzolos projektből, akkor készítsd el újra.

Seller.cs

namespace RealEstateGUI
{
    class Seller
    {
        public int Id { get; set; }
        public string Name { get; set; }
        public string Phone { get; set; }

        public Seller(int id, string name, string phone)
        {
            Id = id;
            Name = name;
            Phone = phone;
        }

        public override string ToString()
        {
            return Name;
        }
    }
}

Fontos:

public override string ToString()
{
    return Name;
}

8. Teljes Form1.cs példa
using MySql.Data.MySqlClient;
using System;
using System.Collections.Generic;
using System.Windows.Forms;

namespace RealEstateGUI
{
    public partial class Form1 : Form
    {
        string connectionString =
            "server=localhost;database=ingatlan;uid=root;pwd=;charset=utf8;";

        List<Seller> sellers = new List<Seller>();

        public Form1()
        {
            InitializeComponent();
            SellersBetoltese();

            listBoxSellers.DataSource = sellers;
            listBoxSellers.DisplayMember = "Name";

            labelSellerName.DataBindings.Add(
                "Text",
                listBoxSellers.DataSource,
                "Name"
            );

            labelSellerPhone.DataBindings.Add(
                "Text",
                listBoxSellers.DataSource,
                "Phone"
            );
        }

        private void SellersBetoltese()
        {
            sellers.Clear();

            using (MySqlConnection connection = new MySqlConnection(connectionString))
            {
                connection.Open();

                string sql = "SELECT id, name, phone FROM sellers";

                using (MySqlCommand command = new MySqlCommand(sql, connection))
                {
                    using (MySqlDataReader reader = command.ExecuteReader())
                    {
                        while (reader.Read())
                        {
                            Seller seller = new Seller(
                                reader.GetInt32("id"),
                                reader.GetString("name"),
                                reader.GetString("phone")
                            );

                            sellers.Add(seller);
                        }
                    }
                }
            }
        }

        private void buttonLoadAds_Click(object sender, EventArgs e)
        {
            if (listBoxSellers.SelectedItem == null)
            {
                MessageBox.Show("Válasszon ki egy eladót!");
                return;
            }

            Seller selectedSeller = (Seller)listBoxSellers.SelectedItem;

            int count = HirdetesekSzama(selectedSeller.Id);

            labelAdCount.Text = count.ToString();
        }

        private int HirdetesekSzama(int sellerId)
        {
            int count = 0;

            using (MySqlConnection connection = new MySqlConnection(connectionString))
            {
                connection.Open();

                string sql = "SELECT COUNT(*) FROM realestates WHERE sellerId = @sellerId";

                using (MySqlCommand command = new MySqlCommand(sql, connection))
                {
                    command.Parameters.AddWithValue("@sellerId", sellerId);

                    count = Convert.ToInt32(command.ExecuteScalar());
                }
            }

            return count;
        }
    }
}

10. Fontos SQL lekérdezések

Egy eladó hirdetéseinek száma
SELECT COUNT(*) 
FROM realestates 
WHERE sellerId = @sellerId;

Ha nálad más az oszlop neve, például seller_id, akkor így:
